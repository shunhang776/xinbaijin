```python
import nox


@nox.session
def lint(session):
    """Ruff 格式和静态检查"""
    session.install("ruff")
    session.run("ruff", "format", "--check", ".")
    session.run("ruff", "check", ".")


@nox.session
def architecture(session):
    """Import Linter 架构合同检查"""
    session.install("import-linter")
    session.run("lint-imports")


@nox.session
def tests(session):
    """pytest 全部测试"""
    session.install(".")
    session.install("pytest", "hypothesis")
    session.run(
        "pytest",
        "--junitxml=artifacts/current/pytest.xml",
        "-v",
    )


@nox.session
def all_checks(session):
    """一键运行所有确定性检查"""
    session.notify("lint")
    session.notify("architecture")
    session.notify("tests")
```

### 3.3.3 编写 OPA Rego 政策

**D:\白槿治理\policy\review.rego**

```rego
package baijin.acceptance

# ===== 证据完整性 =====

deny contains "证据不完整" if {
    input.evidence.complete == false
}

deny contains "证据与当前 commit 不匹配" if {
    input.evidence.commit_matches == false
}

deny contains "存在缺失的报告" if {
    count(input.evidence.missing_reports) > 0
}

# ===== 范围外修改 =====

deny contains "存在范围外文件修改" if {
    input.scope.out_of_scope_files > 0
}

# ===== 确定性检查 =====

deny contains "Ruff 检查失败" if {
    input.checks.ruff != "pass"
}

deny contains "mypy 类型检查失败" if {
    input.checks.mypy != "pass"
}

deny contains "Import Linter 架构合同失败" if {
    input.checks.import_linter != "pass"
}

deny contains "测试失败" if {
    input.checks.pytest_failed > 0
}

deny contains "受保护测试被删除" if {
    input.checks.protected_tests_removed > 0
}

deny contains "密钥泄漏" if {
    input.checks.gitleaks != "pass"
}

deny contains "依赖漏洞" if {
    input.checks.pip_audit != "pass"
}

deny contains "严重安全发现" if {
    input.checks.semgrep_critical > 0
}

# ===== AI 审查结果 =====

deny contains "存在 P0 缺陷" if {
    input.review.p0 > 0
}

deny contains "存在 P1 缺陷" if {
    input.review.p1 > 0
}

deny contains "Codex 拒绝接受" if {
    input.review.safe_to_accept == false
}

# ===== 回滚 =====

deny contains "恢复点缺失" if {
    input.rollback.restore_point_exists == false
}

deny contains "回滚演练失败" if {
    input.rollback.drill_passed == false
}

# ===== 需要人工确认 =====

require_human contains "数据库发生变更" if {
    input.scope.database_changed == true
}

require_human contains "治理政策发生变更" if {
    input.scope.governance_changed == true
}

require_human contains "安全配置发生变更" if {
    input.scope.security_changed == true
}

require_human contains "AI 审查建议人工确认" if {
    input.review.requires_human_review == true
}

# ===== 最终裁决 =====

# 有拒绝项或人工确认项时，不允许自动通过
default allow = false

allow if {
    count(deny) == 0
    count(require_human) == 0
}

# 有人工确认项但无拒绝项时，状态为 NEEDS_HUMAN_REVIEW
needs_human if {
    count(deny) == 0
    count(require_human) > 0
}
```

```python
def run(cmd, **kwargs):
    """运行命令，返回 (returncode, stdout, stderr)"""
    result = subprocess.run(
        cmd, capture_output=True, text=True, shell=True, **kwargs
    )
    return result.returncode, result.stdout, result.stderr


def collect_git_evidence():
    """收集 Git 证据"""
    os.chdir(PROJECT_ROOT)

    # 当前 commit
    rc, stdout, _ = run("git rev-parse HEAD")
    current_commit = stdout.strip() if rc == 0 else "UNKNOWN"

    # 修改文件列表
    rc, stdout, _ = run("git diff --name-status HEAD")
    changed_files = stdout.strip()

    # diff stat
    rc, stdout, _ = run("git diff --stat HEAD")
    diff_stat = stdout.strip()

    # 完整 diff
    rc, stdout, _ = run("git diff --binary HEAD")
    patch = stdout

    # 工作树状态
    rc, stdout, _ = run("git status --porcelain")
    working_tree = stdout.strip()

    # 写入文件
    ARTIFACTS.mkdir(parents=True, exist_ok=True)
    (ARTIFACTS / "current-commit.txt").write_text(current_commit)
    (ARTIFACTS / "changed-files.txt").write_text(changed_files)
    (ARTIFACTS / "diff-stat.txt").write_text(diff_stat)
    (ARTIFACTS / "patch.diff").write_text(patch)
    (ARTIFACTS / "working-tree-status.txt").write_text(working_tree)

    return {
        "current_commit": current_commit,
        "changed_files": changed_files,
    }


def parse_pytest_xml():
    """解析 pytest JUnit XML 得到通过/失败数"""
    xml_path = ARTIFACTS / "pytest.xml"
    if not xml_path.exists():
        return {"passed": 0, "failed": 0, "skipped": 0, "present": False}

    # 简单解析 — 后续可用 xml.etree 增强
    content = xml_path.read_text(encoding="utf-8")
    import re
    testsuite = re.search(r'testsuite.*?tests="(\d+)".*?failures="(\d+)".*?skipped="(\d+)"', content)
    if not testsuite:
        # 尝试匹配多个 testsuite 的汇总
        tests = sum(int(m) for m in re.findall(r'tests="(\d+)"', content))
        failures = sum(int(m) for m in re.findall(r'failures="(\d+)"', content))
        skipped = sum(int(m) for m in re.findall(r'skipped="(\d+)"', content))
        return {"passed": tests - failures - skipped, "failed": failures, "skipped": skipped, "present": True}

    tests = int(testsuite.group(1))
    failures = int(testsuite.group(2))
    skipped = int(testsuite.group(3))
    return {"passed": tests - failures - skipped, "failed": failures, "skipped": skipped, "present": True}

    # 实际使用时根据每个工具的输出文件判断
    return "pass"

    pytest = parse_pytest_xml()

            "commit_matches": True,  # 后续实际比对
            "missing_reports": [],
        },
        "scope": {
            "files_changed": files_changed,
            "files_added": 0,
            "files_deleted": 0,
            "out_of_scope_files": 0,
            "database_changed": False,
            "governance_changed": False,
            "security_changed": False,
        },
        "checks": {
            "ruff": check_tool_output("ruff"),
            "mypy": check_tool_output("mypy"),
            "import_linter": check_tool_output("import_linter"),
            "pytest_passed": pytest["passed"],
            "pytest_failed": pytest["failed"],
            "pytest_skipped": pytest["skipped"],
            "protected_tests_removed": 0,
            "gitleaks": check_tool_output("gitleaks"),
            "pip_audit": check_tool_output("pip_audit"),
            "semgrep_critical": 0,
            "semgrep_high": 0,
        },

            "p1": codex_result.get("p1_count", 0) if codex_result else 0,
            "missing_tests": len(codex_result.get("missing_tests", [])) if codex_result else 0,
            "requires_human_review": codex_result.get("requires_human_review", False) if codex_result else False,
```

```python
def run_cmd(cmd, cwd=None, timeout=120):

def run_conftest():
    """运行 Conftest 政策裁决"""
    review_input = ARTIFACTS / "review-input.json"
    if not review_input.exists():
        return False, "review-input.json 不存在"

    )

def run_codex_review():

    print("[Hook] 调用 Codex CLI 审查...")

def check_repair_cycles():
    """检查修复次数是否超限"""
    status_path = PROJECT_ROOT / "tasks" / "current" / "status.yaml"
    if not status_path.exists():
        return 0
    # 简化解析 — 实际使用 PyYAML
    content = status_path.read_text(encoding="utf-8")
    for line in content.split("\n"):

    return 0

- 人工确认项：{result.get('human_count', 0)}

```
{json.dumps(result, indent=2, ensure_ascii=False)}
```
"""

        print(f"[Hook] Codex 审查完成: safe_to_accept={codex_result.get('safe_to_accept')}")

    # 5. Conftest 最终裁决
    policy_ok, policy_out, policy_err = run_conftest()
    
    # 6. 读取 OPA 结果
    result = {
        "status": "ACCEPTED" if policy_ok else "POLICY_FAILED",
        "allow": policy_ok,
        "deny_count": 0 if policy_ok else 1,
        "human_count": 0,
    }
    generate_final_report(result)

    if policy_ok:
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
    else:
        print(json.dumps({
            "decision": "block",
            "reason": f"政策门禁未通过。请读取 {ARTIFACTS / 'final-report.md'} 后继续修复。"
        }))
        sys.exit(1)


if __name__ == "__main__":
    main()
```
